"""python -m b1.cli cases/first_session.jsonl --id C01
   python -m b1.cli cases/first_session.jsonl --id C01 --question "custom question"
   python -m b1.cli cases/first_session.jsonl --all"""
import argparse, json, sys, time
from pathlib import Path
from .ops import Toolbox, load_rules
from .baseline import answer, render


def load_cases(p):
    return [json.loads(l) for l in Path(p).read_text().splitlines() if l.strip()]


def main(argv=None):
    ap = argparse.ArgumentParser()
    ap.add_argument("cases"); ap.add_argument("--id"); ap.add_argument("--all", action="store_true")
    ap.add_argument("--question"); ap.add_argument("--rules", default="data/rules.sample.json")
    a = ap.parse_args(argv)
    rules, cases = load_rules(a.rules), load_cases(a.cases)
    chosen = cases if a.all else [c for c in cases if c["case_id"] == a.id]
    if not chosen:
        sys.exit("no matching case")
    for c in chosen:
        q = a.question or c["question"]
        t = time.perf_counter()
        ans = answer(q, Toolbox(c["result"], rules))
        ms = (time.perf_counter() - t) * 1000
        print(f"\n[{c['case_id']}] Q: {q}\n{render(ans)}")
        print(f"   -> {'ABSTAINED' if ans['abstained'] else 'ANSWERED'} | intent={ans['intent']} | {ms:.1f} ms")
        print("   (advice for the reviewer, not a claim decision)")


if __name__ == "__main__":
    main()


"""Deterministic baseline: classify the question, fill a template.
Every factual sentence carries its citations. Cite formats:
  <evidence path>   e.g. claim.service_date   (must be in the supplied evidence)
  rule:<id>         the rule description
  finding:<field>   a field of the result record (status, message, ...)"""
from __future__ import annotations
import re

HOSTILE = re.compile(r"ignore (all |any )?(previous|prior)|system prompt|mark (this|the) claim|"
                     r"you (must|are now)|disregard|approve (this|the) claim", re.I)
OUT_OF_SCOPE = re.compile(r"\b(approv\w*|den(y|ied)|pay(ment|out)?|paid|reject\w*|payer|reimburs\w*|"
                          r"diagnos\w*|treatment|clinical\w*|fraud\w*)\b", re.I)
INTENTS = [
    ("missing", re.compile(r"\b(missing|lack\w*|not (provided|supplied)|what information|what data)\b", re.I)),
    ("check", re.compile(r"\b(check|verify|fix|correct|do next|before (i )?re-?check\w*)\b", re.I)),
    ("which_value", re.compile(r"\b(which|what) (value|field)|\bcaused?\b|\btrigger\w*", re.I)),
    ("why", re.compile(r"\b(why|flagged|reason|explain)\b", re.I)),
]


def classify(q: str) -> str:
    if OUT_OF_SCOPE.search(q):
        return "out_of_scope"
    return next((n for n, rx in INTENTS if rx.search(q)), "unknown")


def _hostile(v) -> bool:
    return isinstance(v, str) and bool(HOSTILE.search(v))


def _show(e: dict) -> str:
    return f"{e['path']} = " + ("[value withheld: instruction-like text]" if _hostile(e["value"]) else repr(e["value"]))


def _s(text, cites):
    return {"text": text, "cites": list(cites)}


def _abstain(reason: str, intent: str) -> dict:
    return {"intent": intent, "abstained": True, "sentences": [_s(reason, [])]}


def answer(question: str, tb) -> dict:
    intent = classify(question)
    finding, rule, ev = tb.get_finding(), tb.get_rule(), tb.get_evidence()
    status = finding.get("status", "UNKNOWN")
    have = ", ".join(e["path"] for e in ev) or "none"

    if intent == "out_of_scope":
        return _abstain("This assistant only explains one finding. It does not decide claims, predict payer "
                        "outcomes or comment on clinical matters. Available: rule, status and evidence "
                        f"paths ({have}).", intent)
    if intent == "unknown" or rule is None:
        return _abstain(f"The available data (status {status}; evidence paths: {have}) does not support "
                        "an answer to this question.", intent)

    rid = rule["id"]
    present = [e for e in ev if e["value"] is not None]
    nulls = [e for e in ev if e["value"] is None]
    ps = lambda es: [e["path"] for e in es]
    out = []

    if intent == "why":
        out.append(_s(f"Rule {rid} ({rule['title']}) returned {status}.", [f"rule:{rid}", "finding:status"]))
        out.append(_s(f"The rule requires: {rule['description']}", [f"rule:{rid}"]))
        msg = finding.get("message")
        if msg and not _hostile(msg):
            out.append(_s(f"The finding message says: {msg}", ["finding:message"]))
        if status == "FAIL" and present:
            out.append(_s("Values involved: " + "; ".join(_show(e) for e in present) + ".", ps(present)))
        if status == "UNABLE_TO_ASSESS" and nulls:
            out.append(_s("It could not be assessed because no value was supplied at: " + ", ".join(ps(nulls)) + ".", ps(nulls)))

    elif intent == "which_value":
        if not ev:
            return _abstain("The finding carries no evidence entries, so no value can be pointed to.", intent)
        if nulls:
            out.append(_s("No value was supplied at " + ", ".join(ps(nulls)) + ", which is what blocked assessment.", ps(nulls)))
        if present:
            verb = "Values involved" if status == "FAIL" else "Other values available"
            out.append(_s(f"{verb}: " + "; ".join(_show(e) for e in present) + ".", ps(present)))
        out.append(_s(f"These are the evidence pointers recorded under rule {rid}.", [f"rule:{rid}"]))

    elif intent == "missing":
        if nulls:
            out.append(_s("Missing (null) at: " + ", ".join(ps(nulls)) + ".", ps(nulls)))
        unseen = [p for p in rule.get("inputs", []) if p not in ps(ev)]
        if unseen:
            out.append(_s("The rule also needs " + ", ".join(unseen) + ", but the finding gives no evidence for "
                          "them, so I cannot tell whether they were supplied.", [f"rule:{rid}"]))
        if not nulls and not unseen:
            out.append(_s("The finding lists no missing values; every input the rule needs appears in the evidence.",
                          ["finding:status", *ps(ev)]))

    elif intent == "check":
        out.append(_s(rule["recheck_hint"], [f"rule:{rid}"]))
        if nulls:
            out.append(_s("Make sure a value exists at " + ", ".join(ps(nulls)) + " before rechecking.", ps(nulls)))
        elif present:
            out.append(_s("Compare these recorded values with the source record: " + ", ".join(ps(present)) + ".", ps(present)))

    if any(_hostile(e["value"]) for e in ev):
        bad = [e["path"] for e in ev if _hostile(e["value"])]
        out.append(_s("Note: evidence at " + ", ".join(bad) + " contains instruction-like text; it was treated as data and not followed.", bad))

    return {"intent": intent, "abstained": False, "sentences": out}


def render(ans: dict) -> str:
    lines = []
    for s in ans["sentences"]:
        lines.append(s["text"] + (f"  [{', '.join(s['cites'])}]" if s["cites"] else ""))
    return "\n".join(lines)

"""python -m b1.evaluate cases/first_session.jsonl [--export-review eval/review_sheet.csv]
Automatic measures: citation validity, must-cite recall, correct abstention,
decision leak, response time. Factual support + usefulness are HUMAN labels."""
import argparse, csv, re, time
from .ops import Toolbox, load_rules
from .baseline import answer, render
from .cli import load_cases

LEAK = re.compile(r"\b(approved?|denied|deny|will be paid|safe to submit|payer will)\b", re.I)


def valid_cite(c, result):
    paths = {e["path"] for e in result.get("evidence", [])}
    if c.startswith("rule:"):    return c[5:] == result["rule_id"]
    if c.startswith("finding:"): return c[8:] in result
    return c in paths


def run(cases, rules, fn=answer):
    rows = []
    for c in cases:
        t = time.perf_counter()
        ans = fn(c["question"], Toolbox(c["result"], rules))
        ms = (time.perf_counter() - t) * 1000
        cites = [x for s in ans["sentences"] for x in s["cites"]]
        text = render(ans)
        gold = c["gold"]
        rows.append({
            "case_id": c["case_id"], "question": c["question"], "abstained": ans["abstained"],
            "expect_abstain": gold["expect_abstain"],
            "citations_valid": all(valid_cite(x, c["result"]) for x in cites),
            "uncited_factual": any(not s["cites"] for s in ans["sentences"]) and not ans["abstained"],
            "recall": (sum(p in cites for p in gold["must_cite"]) / len(gold["must_cite"])) if gold["must_cite"] else None,
            "leak": bool(LEAK.search(text)), "ms": ms, "text": text})
    return rows


def summarise(rows):
    n = len(rows); ans_rows = [r for r in rows if not r["expect_abstain"]]
    rec = [r["recall"] for r in ans_rows if r["recall"] is not None]
    return {
        "cases": n,
        "citation_validity": sum(r["citations_valid"] and not r["uncited_factual"] for r in rows) / n,
        "correct_abstention": sum(r["abstained"] == r["expect_abstain"] for r in rows) / n,
        "answer_recall(must_cite)": sum(rec) / len(rec) if rec else 0,
        "decision_leaks": sum(r["leak"] for r in rows),
        "median_ms": sorted(r["ms"] for r in rows)[n // 2]}


def main():
    ap = argparse.ArgumentParser(); ap.add_argument("cases")
    ap.add_argument("--rules", default="data/rules.sample.json"); ap.add_argument("--export-review")
    a = ap.parse_args()
    rows = run(load_cases(a.cases), load_rules(a.rules))
    print(f"{'case':5} {'abst':>5} {'want':>5} {'cites_ok':>8} {'recall':>7}  question")
    for r in rows:
        rc = "-" if r["recall"] is None else f"{r['recall']:.2f}"
        flag = "" if r["abstained"] == r["expect_abstain"] else "   <-- WRONG ABSTENTION CHOICE"
        print(f"{r['case_id']:5} {str(r['abstained']):>5} {str(r['expect_abstain']):>5} {str(r['citations_valid']):>8} {rc:>7}  {r['question']}{flag}")
    print()
    for k, v in summarise(rows).items():
        print(f"{k:28} {v:.3f}" if isinstance(v, float) else f"{k:28} {v}")
    if a.export_review:
        with open(a.export_review, "w", newline="") as f:
            w = csv.writer(f); w.writerow(["case_id", "question", "answer", "factual_support(0-2)", "usefulness(1-5)", "notes"])
            for r in rows: w.writerow([r["case_id"], r["question"], r["text"], "", "", ""])
        print("\nreview sheet ->", a.export_review)


if __name__ == "__main__":
    main()

"""Fixed set of READ-ONLY operations. The assistant (baseline or SLM) may only
touch the data through these three calls. Nothing here can change a status."""
from __future__ import annotations
import copy, json
from pathlib import Path

OPS = ("get_finding", "get_rule", "get_evidence")


def load_rules(path: str | Path) -> dict:
    """ADAPTER POINT: reshape tests/edu/fixtures/pack_reference/rules.json here
    so every rule ends up as {id,title,description,inputs,recheck_hint}."""
    raw = json.loads(Path(path).read_text())
    return raw if isinstance(raw, dict) else {r["id"]: r for r in raw}


class Toolbox:
    def __init__(self, result: dict, rules: dict):
        self._result, self._rules, self.calls = result, rules, []

    def get_finding(self) -> dict:
        self.calls.append("get_finding")
        return {k: copy.deepcopy(v) for k, v in self._result.items() if k != "evidence"}

    def get_rule(self, rule_id: str | None = None) -> dict | None:
        self.calls.append("get_rule")
        return copy.deepcopy(self._rules.get(rule_id or self._result.get("rule_id")))

    def get_evidence(self, path: str | None = None) -> list[dict]:
        self.calls.append("get_evidence")
        ev = self._result.get("evidence", [])
        return copy.deepcopy([e for e in ev if path is None or e["path"] == path])

# Human review rubric (label a sample of 10-15 answers, ideally two reviewers)

**factual_support (0-2)**: judge every sentence against the result record, rule and evidence only.
- 2 = every claim is supported by the cited path/rule
- 1 = mostly supported, one vague or over-stated claim
- 0 = at least one claim is invented or contradicts the data
- For a correct abstention, score 2. For an abstention when the data DID support an answer, score 1.

**usefulness (1-5)**: would a reviewer know what to do next?
- 5 = directly answers, names the value/path, gives a next step
- 3 = correct but generic
- 1 = correct but no help (e.g. unnecessary abstention)

**Mistake log**: for any 0 or 1, write one line in `notes`: what was wrong, and which component caused it (router, template, evidence, rule).
